# Commands carried by the akmesh binary, one per line.
# Used by scripts/build-bb.sh. A name must exist as ./cmds/core/<name> in the
# pinned u-root checkout, and it must build for the target platform.

# shell

# binary formats, read on every platform
readelf

# files and text
# head, find, grep, chmod, cp, xargs, tar, hexdump, base64, wc, tr, uniq, comm and touch are not here: they are forked into internal/forked/, and a fork
# takes the name away from u-root (see scripts/fork-uroot.sh, OURS_ALL).
cat more tail tee cmp
mv rm mkdir ln mktemp readlink realpath

# archives, hashes, encoding
cpio gzip md5sum shasum dd strings

# system and time
date time uptime sleep printenv which man

# filesystems and devices, from u-root; they only build for Linux (acmesh-hdb)
linux: mount umount mknod losetup

# network
ping netcat
# Linux only: netlink and DHCP talk to the kernel, they cannot build for Windows.
linux: ip dhclient
# netstat reads the kernel tables in /proc, which Windows does not have.
linux: netstat

# helpers
yes

# Excluded on Windows (build constraints, see docs/measurements.md):
#   df du free hostname kill ps sync timeout uname
# They stay in the list for Linux builds via scripts/cmds-linux-only.txt.
