# Commands carried by the akmesh binary, one per line.
# Used by scripts/build-bb.sh. A name must exist as ./cmds/core/<name> in the
# pinned u-root checkout, and it must build for the target platform.

# shell

# files and text
# head, find, grep, chmod, cp, xargs, tar, hexdump, base64, wc, tr and touch are not here: they are forked into internal/forked/, and a fork
# takes the name away from u-root (see scripts/fork-uroot.sh, OURS_ALL).
ls cat more tail uniq tee cmp comm
mv rm mkdir ln mktemp readlink realpath

# archives, hashes, encoding
cpio gzip md5sum shasum dd strings

# system and time
date time uptime sleep id printenv which man

# network
ping netcat

# helpers
seq yes

#   df du free hostname kill ps sync timeout uname
