# Commands carried by the acmesh binary, one per line.
# Used by scripts/build-bb.sh. A name must exist as ./cmds/core/<name> in the
# pinned u-root checkout, and it must build for the target platform.

# shell

# files and text
ls cat more head tail grep find sort uniq tr tee wc cmp comm
cp mv rm mkdir touch ln chmod mktemp readlink realpath

# archives, hashes, encoding
tar cpio gzip base64 md5sum shasum dd strings hexdump

# system and time
date time uptime sleep id printenv which man

# network
ping netcat

# helpers
seq yes xargs
